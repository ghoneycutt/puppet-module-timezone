class timezone (
  String[1] $timezone = 'UTC',
) {

  $tzdata = "/usr/share/zoneinfo/${timezone}"

  file { '/etc/localtime':
    ensure => link,
    target => $tzdata,
  }

  if $facts['os']['family'] == 'Debian' {
    file { '/etc/timezone':
      owner   => 'root',
      group   => 'root',
      mode    => '0644',
      content => "${timezone}\n",
    }
  }
}
